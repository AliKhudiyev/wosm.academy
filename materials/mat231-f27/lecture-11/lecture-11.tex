\begin{frame}{Last time}
  \begin{itemize}
    \item Partial derivatives $\frac{\partial f}{\partial x_i}$: differentiate in one variable,
      freeze the others.
    \item $f$ is \textbf{differentiable} at $\va$ if
      $f(\va + \vh) = f(\va) + Df(\va)\,\vh + o(\norm{\vh})$ for a linear map $Df(\va)$.
    \item Then $Df(\va)$ is the matrix of partial derivatives; for $f\colon \R^2 \to \R$ it gives
      the tangent plane.
    \item Continuous partial derivatives ($C^1$) imply differentiability.
  \end{itemize}
  \vskip0.6em
  Today: for a real-valued $f$, the row $Df(\va)$ becomes a \emph{vector}, the gradient. It
  collects all first-order information about $f$ near $\va$, and it points uphill.
\end{frame}

\section{Directional derivatives}

\begin{frame}{The rate of change in a direction}
  \begin{definition}
    Let $f\colon U \to \R$, $U \subseteq \R^n$ open, $\va \in U$, and $\vv$ a \textbf{unit} vector.
    The \textbf{directional derivative} of $f$ at $\va$ in the direction $\vv$ is
    \[ D_\vv f(\va) = \frac{\mathrm{d}}{\mathrm{d}t}\, f(\va + t\vv)\Big|_{t=0}
       = \lim_{t \to 0} \frac{f(\va + t\vv) - f(\va)}{t} , \]
    if the limit exists.
  \end{definition}
  \pause
  \begin{itemize}
    \item Walk from $\va$ along the line through $\va$ in direction $\vv$: $D_\vv f(\va)$ is how fast
      $f$ changes, per unit of distance.
    \item The partial derivatives are the special cases $\vv = \mathbf{e}_i$:
      $D_{\mathbf{e}_i} f = \frac{\partial f}{\partial x_i}$.
    \item $\vv$ must be a unit vector; otherwise the rate is scaled by $\norm{\vv}$.
  \end{itemize}
\end{frame}

\begin{frame}{Directional derivatives from the derivative}
  \begin{theorem}
    If $f$ is differentiable at $\va$, then for every unit vector $\vv$:
    \quad $D_\vv f(\va) = Df(\va)\,\vv = \sum_{i=1}^n \frac{\partial f}{\partial x_i}(\va)\, v_i$.
  \end{theorem}
  \pause
  \begin{proof}
    With $\vh = t\vv$ and $\norm{t\vv} = |t|$:
    $f(\va + t\vv) = f(\va) + t\, Df(\va)\,\vv + o(|t|)$. Subtract $f(\va)$, divide by $t$, let $t \to 0$.
  \end{proof}
  \pause
  So one vector, $Df(\va)$, gives the rate of change in every direction.
\end{frame}

\begin{frame}{Directional derivatives are not enough}
  \begin{warning}
    The converse fails. For $f = \frac{x^2 y}{x^4 + y^2}$, $f(0,0) = 0$ (Lecture 9), every
    directional derivative at $(0, 0)$ exists: $D_\vv f = \frac{v_1^2}{v_2}$ for $v_2 \ne 0$, and $0$
    otherwise. Yet $f$ is not even continuous there --- and $\vv \mapsto D_\vv f$ is not linear.
  \end{warning}
  As with partial derivatives (Lecture 10): knowing $f$ along every line through $\va$ does not
  control $f$ near $\va$. Differentiability is the condition that does.
\end{frame}

\begin{frame}{Examples}
  \begin{example}
    $f(x, y) = x^2 y + y^3$ at $(1, 2)$, in the direction of $(3, 4)$.
    The unit vector is $\vv = \frac15(3, 4)$; \ $f_x = 2xy = 4$, \ $f_y = x^2 + 3y^2 = 13$; so
    $D_\vv f(1, 2) = 4 \cdot \frac35 + 13 \cdot \frac45 = \frac{64}{5} = 12.8$.
  \end{example}
  \pause
  \begin{example}
    $f(x, y, z) = xyz$ at $(1, 1, 1)$, in the direction $\frac{1}{\sqrt3}(1, 1, 1)$:
    \ all partial derivatives are $1$, so $D_\vv f = \frac{3}{\sqrt3} = \sqrt3$.
  \end{example}
  {\small Interpretation: walking from $(1, 2)$ towards $(4, 6)$, the graph of $f$ rises
  $12.8$ units per unit of horizontal distance.\par}
\end{frame}

\section{The gradient vector}

\begin{frame}{The gradient}
  \begin{definition}
    The \textbf{gradient} of $f$ at $\va$ is the column vector of partial derivatives,
    $\nabla f(\va) = \bigl(\frac{\partial f}{\partial x_1}(\va), \dots,
    \frac{\partial f}{\partial x_n}(\va)\bigr)^{\!\top}$.
  \end{definition}
  \begin{itemize}
    \item $Df(\va) = \nabla f(\va)^\top$: the derivative is a row, the gradient its transpose.
    \item So for differentiable $f$: \ $D_\vv f(\va) = \nabla f(\va) \cdot \vv$, and
      $f(\va + \vh) = f(\va) + \nabla f(\va) \cdot \vh + o(\norm{\vh})$.
  \end{itemize}
  \pause
  \begin{example}
    $\nabla(x^2 + y^2) = (2x, 2y)$, \quad
    $\nabla e^{xy} = (y\, e^{xy}, \ x\, e^{xy})$, \quad
    $\nabla(xyz) = (yz, xz, xy)$.
  \end{example}
\end{frame}

\begin{frame}{The direction of steepest ascent}
  \begin{theorem}
    Let $f$ be differentiable at $\va$ with $\nabla f(\va) \ne \mathbf{0}$. Among all unit vectors
    $\vv$, \ $D_\vv f(\va)$ is largest for $\vv = \frac{\nabla f(\va)}{\norm{\nabla f(\va)}}$, where it
    equals $\norm{\nabla f(\va)}$, and smallest in the opposite direction.
  \end{theorem}
  \pause
  \begin{proof}
    By the Cauchy--Schwarz inequality (Linear Algebra, Unit 13),
    $|D_\vv f(\va)| = |\nabla f(\va) \cdot \vv| \le \norm{\nabla f(\va)}\,\norm{\vv} = \norm{\nabla f(\va)}$,
    with equality exactly when $\vv$ is a multiple of $\nabla f(\va)$.
  \end{proof}
  \pause
  Equivalently $D_\vv f(\va) = \norm{\nabla f(\va)} \cos\theta$, with $\theta$ the angle between $\vv$
  and $\nabla f(\va)$; it is $0$ perpendicular to $\nabla f(\va)$.
\end{frame}

\begin{frame}{Steepest descent}
  \begin{example}[A hiker]
    The height of a hill is $f(x, y) = 100 - x^2 - 2y^2$. At $(3, 2)$:
    $\nabla f = (-2x, -4y) = (-6, -8)$, with $\norm{\nabla f} = 10$.
    Steepest way up: the direction $(-0.6, -0.8)$, at slope $10$. Steepest way down:
    $(0.6, 0.8)$, at slope $-10$. Level walking: $\pm(0.8, -0.6)$, perpendicular to $\nabla f$.
  \end{example}
  \pause
  \begin{block}{Gradient descent (Lecture 15)}
    Step against the gradient: $\vx_{k+1} = \vx_k - \eta\, \nabla f(\vx_k)$, with a small step
    size $\eta > 0$. This simple rule trains most machine-learning models.
  \end{block}
\end{frame}

\begin{frame}{The gradient is perpendicular to level sets}
  \begin{theorem}
    Let $f$ be differentiable, and $\vr(t)$ a differentiable curve in the level set
    $\{f = c\}$ with $\vr(0) = \va$. Then $\nabla f(\va) \cdot \vr'(0) = 0$.
  \end{theorem}
  \pause
  \begin{proof}
    $f(\vr(s)) = c$ for all $s$, and by differentiability
    $0 = f(\vr(s)) - f(\va) = \nabla f(\va) \cdot (\vr(s) - \va) + o(\norm{\vr(s) - \va})$.
    Divide by $s$, let $s \to 0$ ($\frac{\norm{\vr(s) - \va}}{|s|}$ stays bounded):
    $0 = \nabla f(\va) \cdot \vr'(0)$. A first case of the chain rule (Lecture 12).
  \end{proof}
  \pause
  So $\nabla f(\va)$ is perpendicular to every curve in the level set through $\va$: it is a
  \textbf{normal vector} to the level set.
\end{frame}

\begin{frame}{Tangent lines and tangent planes}
  \begin{columns}[c]
    \begin{column}{0.56\textwidth}
      \begin{itemize}
        \item Level curve, $\nabla f(\va) \ne \mathbf 0$: the tangent line at $\va$ is
          $\nabla f(\va) \cdot (\vx - \va) = 0$.
        \item Level surface: the tangent plane at $\va$ is
          $\nabla f(\va) \cdot (\vx - \va) = 0$.
      \end{itemize}
      \pause
      \begin{example}[Ellipse $x^2 + 4y^2 = 8$ at $(2, 1)$]
        $\nabla f = (2x, 8y) = (4, 8)$: \ $4(x - 2) + 8(y - 1) = 0$, that is $x + 2y = 4$.
      \end{example}
    \end{column}

\end{frame}

\begin{frame}{Tangent planes to level surfaces}
  \begin{example}[Sphere $x^2 + y^2 + z^2 = 9$ at $(1, 2, 2)$]
    $\nabla f = (2x, 2y, 2z) = (2, 4, 4)$: \ $2(x - 1) + 4(y - 2) + 4(z - 2) = 0$, that is
    $x + 2y + 2z = 9$. The normal points away from the centre, as it should.
  \end{example}
  \pause
  \begin{example}[Graphs are level sets]
    The graph $z = g(x, y)$ is the level set $F = 0$ of $F(x, y, z) = g(x, y) - z$, with
    $\nabla F = (g_x, g_y, -1)$. The tangent plane
    $g_x(a, b)(x - a) + g_y(a, b)(y - b) - (z - g(a, b)) = 0$ is the one from Lecture 10.
  \end{example}
\end{frame}

\begin{frame}{Where the gradient vanishes}
  \begin{itemize}
    \item At a summit, every direction goes down: $D_\vv f(\va) \le 0$ for all $\vv$, and also for
      $-\vv$. So $\nabla f(\va) \cdot \vv = 0$ for all $\vv$, that is $\nabla f(\va) = \mathbf 0$.
    \item At a saddle point (a mountain pass), the gradient is also $\mathbf 0$, but the function
      goes up in some directions and down in others.
    \item Near a point where $\nabla f \ne \mathbf 0$, the contours look like nearly parallel,
      evenly spaced lines.
  \end{itemize}
  \pause
  \begin{remark}
    Points with $\nabla f = \mathbf 0$ are called \textbf{critical points}. Telling summits, pits and
    saddles apart needs second derivatives: Lectures 13 and 14.
  \end{remark}
\end{frame}

\section{Rules for the gradient}

\begin{frame}{Rules}
  For differentiable $f, g$ and constants $\alpha, \beta$:
  \begin{align*}
    \nabla(\alpha f + \beta g) &= \alpha \nabla f + \beta \nabla g, &
    \nabla(fg) &= f\, \nabla g + g\, \nabla f, \\
    \nabla\Bigl(\frac fg\Bigr) &= \frac{g\, \nabla f - f\, \nabla g}{g^2} \quad (g \ne 0), &
    \nabla\bigl(\varphi(f)\bigr) &= \varphi'(f)\, \nabla f .
  \end{align*}
  {\small Each follows from the one-variable rule applied to each partial derivative. The last is
  a case of the chain rule (Lecture 12).\par}
  \pause
  \begin{example}
    $\nabla \norm{\vx} = \nabla \sqrt{\norm{\vx}^2} = \frac{1}{2\norm{\vx}} \nabla \norm{\vx}^2
    = \frac{\vx}{\norm{\vx}}$ \ for $\vx \ne \mathbf 0$: a unit vector pointing away from $\mathbf 0$.
  \end{example}
\end{frame}

\begin{frame}{Gradients in vector notation}
  For $\vx \in \R^n$, a fixed vector $\va$ and a fixed $n \times n$ matrix $A$:
  \[ \nabla(\va \cdot \vx) = \va, \qquad
     \nabla\bigl(\norm{\vx}^2\bigr) = 2\vx, \qquad
     \nabla\bigl(\vx^\top A \vx\bigr) = (A + A^\top)\vx . \]
  \pause
  \begin{proof}[Proof of the last one]
    $\vx^\top A \vx = \sum_{i,j} a_{ij} x_i x_j$. Differentiating in $x_k$, the terms with $i = k$ give
    $\sum_j a_{kj} x_j = (A\vx)_k$ and those with $j = k$ give $\sum_i a_{ik} x_i = (A^\top \vx)_k$.
  \end{proof}
  \pause
  \begin{itemize}
    \item For symmetric $A$: \ $\nabla(\vx^\top A \vx) = 2A\vx$ --- like $(a x^2)' = 2ax$.
    \item These rules are used constantly in machine learning: they turn ``differentiate a loss''
      into a few lines of matrix algebra.
  \end{itemize}
\end{frame}

\section{Application: least squares}

\begin{frame}{The normal equations}
  Given an $m \times n$ matrix $A$ and $\vb \in \R^m$, find $\vx$ minimising
  $E(\vx) = \norm{A\vx - \vb}^2$.
  \pause
  \[ E(\vx) = (A\vx - \vb)^\top (A\vx - \vb)
     = \vx^\top A^\top A \vx - 2\, (A^\top \vb) \cdot \vx + \vb^\top \vb . \]
  \pause
  $A^\top A$ is symmetric, so by the rules:
  \[ \nabla E(\vx) = 2A^\top A \vx - 2A^\top \vb . \]
  \pause
  At a minimum the gradient is $\mathbf 0$:
  \[ A^\top A\, \vx = A^\top \vb \qquad \text{(the normal equations).} \]
  {\small That every solution is a minimum follows from the second derivative $2A^\top A$
  (Lectures 13--14).\par}
\end{frame}

\begin{frame}{Two routes to one equation}
  \begin{itemize}
    \item \textbf{Linear Algebra} (Units 14 and 16): $A\vx$ is closest to $\vb$ when it is the
      orthogonal projection of $\vb$ onto the column space of $A$. Then the residual $\vb - A\vx$ is
      perpendicular to every column: $A^\top(\vb - A\vx) = \mathbf 0$.
    \item \textbf{Calculus}: set the gradient of $\norm{A\vx - \vb}^2$ to zero.
  \end{itemize}
  \pause
  \vskip0.3em
  The same equation $A^\top A\vx = A^\top \vb$. Geometry explains \emph{why} the answer is a
  projection; calculus generalises to losses that are not quadratic, where no projection exists ---
  and there we use gradient descent.
\end{frame}

\section{Summary and practice}

\begin{frame}{Summary}
  \begin{itemize}
    \item \textbf{Directional derivative}: $D_\vv f(\va) = \frac{\mathrm d}{\mathrm dt} f(\va + t\vv)|_{t=0}$,
      for a unit vector $\vv$; the slope of the graph in direction $\vv$.
    \item \textbf{Gradient} $\nabla f$: the vector of partial derivatives. For differentiable $f$,
      $D_\vv f = \nabla f \cdot \vv$.
    \item $\nabla f$ points in the direction of steepest ascent, with rate $\norm{\nabla f}$;
      $-\nabla f$ is steepest descent (gradient descent).
    \item $\nabla f$ is perpendicular to level sets; tangent lines and planes:
      $\nabla f(\va) \cdot (\vx - \va) = 0$.
    \item Rules: $\nabla(\va \cdot \vx) = \va$, \ $\nabla \norm{\vx}^2 = 2\vx$, \
      $\nabla(\vx^\top A \vx) = (A + A^\top)\vx$.
    \item Least squares: $\nabla \norm{A\vx - \vb}^2 = 0$ gives $A^\top A\vx = A^\top \vb$.
  \end{itemize}
\end{frame}

\begin{frame}{Practice}
  \small
  \begin{enumerate}
    \item Find $D_\vv f$ for $f(x, y) = x e^y$ at $(2, 0)$ in the direction of $(1, -1)$.
    \item Find the direction of steepest ascent of $f(x, y, z) = x^2 y + y z^2$ at $(1, 1, 1)$,
      and the rate of increase in that direction.
    \item Find the tangent line to the curve $x^3 + y^3 = 9$ at $(1, 2)$.
    \item Find the tangent plane to the ellipsoid $x^2 + 2y^2 + 3z^2 = 6$ at $(1, 1, 1)$.
    \item Show that $\nabla \frac{1}{\norm{\vx}} = -\frac{\vx}{\norm{\vx}^3}$ for $\vx \ne \mathbf 0$.
    \item For $A = \begin{pmatrix} 1 & 2 \\ 0 & 3 \end{pmatrix}$, compute $\nabla(\vx^\top A\vx)$ directly
      and with the formula.
    \item Fit a line $y = c_0 + c_1 t$ to $(0, 0)$, $(1, 1)$, $(2, 1)$, $(3, 2)$ by least squares.
    \item On a contour plot, in which directions is $D_\vv f(\va) = 0$? Explain.
  \end{enumerate}
\end{frame}

\begin{frame}{Answers}
  \small
  \begin{enumerate}
    \item $\nabla f = (e^y, x e^y) = (1, 2)$, \ $\vv = \frac{1}{\sqrt2}(1, -1)$: \
      $D_\vv f = -\frac{1}{\sqrt2}$.
    \item $\nabla f = (2xy, \ x^2 + z^2, \ 2yz) = (2, 2, 2)$: direction $\frac{1}{\sqrt3}(1, 1, 1)$,
      rate $2\sqrt3$.
    \item $\nabla = (3x^2, 3y^2) = (3, 12)$: \ $3(x - 1) + 12(y - 2) = 0$, \ $x + 4y = 9$.
    \item $\nabla = (2, 4, 6)$: \ $x + 2y + 3z = 6$.
    \item $\nabla (\norm{\vx}^2)^{-1/2} = -\frac12 (\norm{\vx}^2)^{-3/2} \cdot 2\vx$.
    \item $\vx^\top A\vx = x_1^2 + 2x_1x_2 + 3x_2^2$, so $\nabla = (2x_1 + 2x_2, \ 2x_1 + 6x_2)$;
      and $A + A^\top = \begin{pmatrix} 2 & 2 \\ 2 & 6 \end{pmatrix}$.
    \item $A^\top A = \begin{pmatrix} 4 & 6 \\ 6 & 14 \end{pmatrix}$, \
      $A^\top \vb = \begin{pmatrix} 4 \\ 9 \end{pmatrix}$: \ $y = 0.1 + 0.6\, t$.
    \item Along the contour through $\va$ (perpendicular to $\nabla f(\va)$): there $f$ is
      momentarily constant.
  \end{enumerate}
\end{frame}
