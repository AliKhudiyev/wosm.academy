\section{Critical points}

\begin{frame}{Local and global extrema}
  Let $f\colon U \to \R$ with $U \subseteq \R^n$, and $\va \in U$.
  \begin{definition}
    \begin{itemize}
      \item $\va$ is a \textbf{local minimum} of $f$ if $f(\va) \le f(\vx)$ for all $\vx \in U$
        in some ball $B(\va, r)$.
      \item $\va$ is a \textbf{global minimum} if $f(\va) \le f(\vx)$ for all $\vx \in U$.
      \item \textbf{Strict}: $<$ instead of $\le$, for $\vx \ne \va$. \ Maxima: reverse the
        inequalities.
    \end{itemize}
  \end{definition}
  \pause
  \begin{itemize}
    \item Extrema = minima and maxima. Maximising $f$ is minimising $-f$, so we usually
      speak of minima.
    \item Fitting a model, training a network, choosing a design: each minimises a function of
      many variables (a loss, an error, a cost).
  \end{itemize}
\end{frame}

\begin{frame}{Fermat's theorem}
  \begin{theorem}[Fermat]
    If $f$ has a local extremum at an interior point $\va$ of $U$ and is differentiable at $\va$,
    then $\nabla f(\va) = \mathbf 0$.
  \end{theorem}
  \pause
  \textbf{Proof.} For each $i$, the function $g(t) = f(\va + t\mathbf e_i)$ of one variable has a
  local extremum at $t = 0$, so $0 = g'(0) = f_{x_i}(\va)$ (Calculus~I). \qed
  \pause
  \begin{definition}
    A \textbf{critical point} of $f$ is an interior point $\va$ with $\nabla f(\va) = \mathbf 0$:
    the tangent plane there is horizontal. A critical point that is not a local extremum is a
    \textbf{saddle point}.
  \end{definition}
  {\small Points where $f$ is not differentiable, and boundary points, are checked separately.\par}
\end{frame}

\begin{frame}{The converse is false}
  \begin{columns}[c]
    \begin{column}{0.58\textwidth}
      \begin{example}[$f(x, y) = x^2 - y^2$]
        $\nabla f = (2x, -2y)$ vanishes at $(0, 0)$. Along the $x$-axis $f = x^2$ has a minimum
        there, along the $y$-axis $f = -y^2$ a maximum. A saddle.
      \end{example}
      \pause
      \begin{example}[$f(x, y) = x^3 + y^2$]
        $\nabla f = (3x^2, 2y)$ vanishes at $(0, 0)$, but $f(t, 0) = t^3$ takes both signs
        near $t = 0$. A saddle again, of a different shape.
      \end{example}
    \end{column}
    \begin{column}{0.4\textwidth}
      \centering
      \begin{tikzpicture}
        \begin{axis}[width=5.2cm, height=4.6cm, view={30}{30}, axis lines=none,

    \end{column}
  \end{columns}
\end{frame}

\section{The second-derivative test}

\begin{frame}{The idea}
  At a critical point $\va$ the linear term of Taylor's formula vanishes:
  \[ f(\va + \vh) = f(\va) + \tfrac12\, \vh^\top H \vh + o(\norm{\vh}^2), \qquad H = H_f(\va) . \]
  \pause
  Near $\va$, $f$ looks like its quadratic term. By the spectral theorem (Linear Algebra, Unit 20)
  $H$ has an orthonormal basis of eigenvectors $\vu_1, \dots, \vu_n$ with eigenvalues
  $\lambda_1, \dots, \lambda_n$, and
  \[ \vh = \sum_i c_i \vu_i \quad\Longrightarrow\quad \vh^\top H \vh = \sum_i \lambda_i c_i^2 . \]
  \pause
  \begin{itemize}
    \item All $\lambda_i > 0$: a bowl. All $\lambda_i < 0$: a cap. Mixed signs: a saddle.
    \item Some $\lambda_i = 0$: in those directions the quadratic term is flat, and the error term
      $o(\norm{\vh}^2)$ decides.
  \end{itemize}
\end{frame}

\begin{frame}{The second-derivative test}
  \begin{theorem}
    Let $f$ be $C^2$ near a critical point $\va$, and $H = H_f(\va)$.
    \begin{enumerate}
      \item $H$ positive definite (all $\lambda_i > 0$) $\Rightarrow$ $\va$ is a strict local minimum.
      \item $H$ negative definite (all $\lambda_i < 0$) $\Rightarrow$ $\va$ is a strict local maximum.
      \item $H$ indefinite (eigenvalues of both signs) $\Rightarrow$ $\va$ is a saddle point.
    \end{enumerate}
    In every other case ($\det H = 0$, no eigenvalues of both signs) the test is
    \textbf{inconclusive}.
  \end{theorem}
  \pause
  {\small Compare Calculus~I: $f'(a) = 0$ and $f''(a) > 0$ give a minimum; $f''(a) = 0$ decides
  nothing ($x^3$, $x^4$, $-x^4$).\par}
\end{frame}

\begin{frame}{Proof}
  \textbf{(1)} Let $\lambda > 0$ be the smallest eigenvalue of $H$. Then
  $\vh^\top H\vh = \sum_i \lambda_i c_i^2 \ge \lambda \sum_i c_i^2 = \lambda \norm{\vh}^2$.
  \pause
  Write $f(\va + \vh) - f(\va) = \frac12 \vh^\top H \vh + r(\vh)$ with $r(\vh) = o(\norm{\vh}^2)$.
  For $\norm{\vh}$ small enough, $|r(\vh)| \le \frac{\lambda}{4}\norm{\vh}^2$, so
  \[ f(\va + \vh) - f(\va) \ \ge\ \tfrac{\lambda}{2}\norm{\vh}^2 - \tfrac{\lambda}{4}\norm{\vh}^2
     \ =\ \tfrac{\lambda}{4}\norm{\vh}^2 \ >\ 0 \qquad (\vh \ne \mathbf 0) . \]
  \pause
  \textbf{(2)} Apply (1) to $-f$.
  \pause\\[0.4em]
  \textbf{(3)} Take unit eigenvectors $\vu$, $\vv$ with eigenvalues $\lambda > 0 > \mu$. Then
  \[ f(\va + t\vu) - f(\va) = \tfrac{\lambda}{2} t^2 + o(t^2) > 0, \qquad
     f(\va + t\vv) - f(\va) = \tfrac{\mu}{2} t^2 + o(t^2) < 0 \]
  for small $t \ne 0$: neither a minimum nor a maximum. \qed
\end{frame}

\begin{frame}{Two variables: the discriminant}
  For $n = 2$, \ $H = \begin{pmatrix} f_{xx} & f_{xy} \\ f_{xy} & f_{yy} \end{pmatrix}$ \ and \
  $D = \det H = f_{xx} f_{yy} - f_{xy}^2 = \lambda_1 \lambda_2$.
  \begin{block}{The test for $f(x, y)$ at a critical point}
    \begin{tabular}{@{}ll@{}}
      $D > 0$ and $f_{xx} > 0$ & local minimum \\
      $D > 0$ and $f_{xx} < 0$ & local maximum \\
      $D < 0$                  & saddle point \\
      $D = 0$                  & inconclusive
    \end{tabular}
  \end{block}
  \pause
  {\small Why: $D < 0$ means $\lambda_1, \lambda_2$ have opposite signs. $D > 0$ means the same sign;
  then $f_{xx} f_{yy} > f_{xy}^2 \ge 0$, so $f_{xx}$ and $f_{yy}$ have that sign too, and so does
  $\lambda_1 + \lambda_2 = f_{xx} + f_{yy}$.\par}
  \pause
  {\small In $n$ variables, Sylvester's criterion (Linear Algebra, Unit 21): $H$ is positive definite
  if and only if all its leading principal minors are positive.\par}
\end{frame}

  $\nabla f = (4x^3 - 4y, \ 4y^3 - 4x) = \mathbf 0$ \ gives $y = x^3$ and $x = y^3 = x^9$, so
  $x \in \{0, 1, -1\}$: \ the critical points are $(0, 0)$, $(1, 1)$, $(-1, -1)$.
  \pause
  \[ H = \begin{pmatrix} 12x^2 & -4 \\ -4 & 12y^2 \end{pmatrix} . \]
  \begin{itemize}
    \item $(0, 0)$: $D = -16 < 0$: \ saddle, $f = 1$.
    \item $(\pm 1, \pm 1)$: $D = 144 - 16 = 128 > 0$, $f_{xx} = 12 > 0$: \ local minima, $f = -1$.
  \end{itemize}
  \pause
  \textbf{Global?} \ $x^4 + y^4 \ge \frac12 (x^2 + y^2)^2$ and $4|xy| \le 2(x^2 + y^2)$, so
  $f \to \infty$ as $\norm{(x, y)} \to \infty$. Then $f$ has a global minimum (extreme value theorem on a
  large disc), it is a critical point, and so the global minimum value is $-1$.
\end{frame}

\begin{frame}{When the test is inconclusive}
  At $(0, 0)$, these three functions all have $\nabla f = \mathbf 0$ and
  $H = \begin{pmatrix} 2 & 0 \\ 0 & 0 \end{pmatrix}$, \ $D = 0$:
  \begin{itemize}
    \item $x^2 + y^4 \ge 0 = f(0, 0)$: \ a strict minimum.
    \item $x^2 - y^4$: \ $f(t, 0) = t^2 > 0$, \ $f(0, t) = -t^4 < 0$: \ a saddle.
    \item $x^2 + y^3$: \ $f(0, t) = t^3$ takes both signs: \ a saddle.
  \end{itemize}
  \pause
  \begin{warning}[$D = 0$ decides nothing]
    The quadratic term is flat along the $y$-axis, and the higher-order terms decide. Look at
    $f$ itself: along lines and curves through the point, or by completing squares.
  \end{warning}
\end{frame}

\section{Global extrema}

\begin{frame}{Global extrema on a closed bounded set}
  \textbf{Extreme value theorem} (Lecture 9): a continuous function on a nonempty closed and
  bounded set $K \subseteq \R^n$ has a global minimum and a global maximum on $K$.
  \pause\\[0.4em]
  Where are they? Either inside $K$ --- then, by Fermat, at a critical point --- or on the boundary.
  \begin{block}{Recipe}
    \begin{enumerate}
      \item Find the critical points in the interior of $K$.
      \item Find the extreme values on the boundary: parametrise it (today), or use Lagrange
        multipliers (Lecture 16).
      \item Compare all the values found. No second-derivative test is needed.
    \end{enumerate}
  \end{block}
\end{frame}

\begin{frame}{Example}
  \begin{example}[$f(x, y) = x^2 + y^2 - 2x$ on the disc $x^2 + y^2 \le 4$]
    \textbf{Interior.} $\nabla f = (2x - 2, \ 2y) = \mathbf 0$ at $(1, 0)$, where $f = -1$.
    \pause\\[0.3em]
    \textbf{Boundary.} On $x^2 + y^2 = 4$: \ $f = 4 - 2x$ with $-2 \le x \le 2$. Smallest value $0$ at
    $(2, 0)$, largest $8$ at $(-2, 0)$.
    \pause\\[0.3em]
    \textbf{Compare} $-1$, $0$, $8$: \ the global minimum is $-1$ at $(1, 0)$, the global maximum
    $8$ at $(-2, 0)$.
  \end{example}
  \pause
  {\small Check: $f = (x - 1)^2 + y^2 - 1$ is the squared distance from $(1, 0)$, minus $1$. It is
  smallest at $(1, 0)$ and largest at the point of the disc farthest from $(1, 0)$.\par}
\end{frame}

\begin{frame}{One critical point is not enough}
  In one variable: if a differentiable $f\colon \R \to \R$ has only one critical point and it is a
  local minimum, it is the global minimum. \textbf{Not in two variables.}
  \pause
  \begin{example}[$f(x, y) = x^2 + y^2 (1 - x)^3$]
    $\nabla f = \bigl(2x - 3y^2(1 - x)^2, \ 2y(1 - x)^3\bigr)$. The second component vanishes if
    $y = 0$ or $x = 1$; then the first gives $x = 0$, or $2 = 0$. So $(0, 0)$ is the only
    critical point.
    \pause\\[0.3em]
    There $H = \begin{pmatrix} 2 & 0 \\ 0 & 2 \end{pmatrix}$: a strict local minimum, with $f = 0$.
    \pause\\[0.3em]
    But $f(4, 1) = 16 - 27 = -11 < 0$.
  \end{example}
  {\small Moral: a local minimum becomes global only with an extra argument --- growth at infinity,
  as for $x^4 + y^4 - 4xy + 1$, or convexity.\par}
\end{frame}

\section{Convexity}

\begin{frame}{Convex sets and convex functions}
  \begin{columns}[c]
    \begin{column}{0.6\textwidth}
      \begin{definition}
        \begin{itemize}
          \item $C \subseteq \R^n$ is \textbf{convex} if it contains the segment between any two
            of its points.
          \item $f\colon C \to \R$ on a convex set is \textbf{convex} if for all $\vx, \vy \in C$
            and $0 \le t \le 1$:
            \[ f\bigl((1 - t)\vx + t\vy\bigr) \le (1 - t) f(\vx) + t f(\vy) . \]
          \item \textbf{Strictly convex}: $<$ whenever $\vx \ne \vy$ and $0 < t < 1$.
          \item $f$ is \textbf{concave} if $-f$ is convex.
        \end{itemize}
      \end{definition}
    \end{column}
    \begin{column}{0.38\textwidth}
      \centering
      \begin{tikzpicture}
        \begin{axis}[wosm axis, width=\linewidth, height=4.4cm, xmin=-1.6, xmax=2.2,
            ymin=-0.2, ymax=3.6, xtick=\empty, ytick=\empty]

          \addplot[wosm-brick, thick, mark=*, mark size=1.5pt] coordinates {(-1.2, 0.78) (1.6, 1.90)};
        \end{axis}

      \par\small The chord lies above the graph. \\ Along every line through $C$, $f$ is convex.
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Tests for convexity}
  \begin{theorem}
    Let $C$ be open and convex.
    \begin{enumerate}
      \item If $f$ is $C^1$: \ $f$ is convex $\iff$ \
        $f(\vy) \ge f(\vx) + \nabla f(\vx) \cdot (\vy - \vx)$ \ for all $\vx, \vy \in C$.
      \item If $f$ is $C^2$: \ $f$ is convex $\iff$ $H_f(\vx)$ is positive semidefinite for every
        $\vx \in C$. \ If $H_f(\vx)$ is positive definite for every $\vx$, $f$ is strictly convex.
    \end{enumerate}
  \end{theorem}
  \begin{itemize}
    \item (1): the graph lies above every tangent plane.
    \item (2): the converse of the last statement fails --- $x^4$ is strictly convex, but
      $f''(0) = 0$.
  \end{itemize}
\end{frame}

\begin{frame}{Why the tests work, and how to use them}
  \begin{proof}[Idea]
    Restrict to lines: $f$ is convex if and only if every $g(t) = f(\vx + t(\vy - \vx))$ is, and
    $g''(t) = (\vy - \vx)^\top H_f\, (\vy - \vx)$ by the chain rule. Now use the one-variable facts
    (Calculus~I, Unit 14).
  \end{proof}
  \pause
  \textbf{Examples.}
  \begin{itemize}
    \item $x^2 + xy + y^2$: \ $H = \left(\begin{smallmatrix} 2 & 1 \\ 1 & 2 \end{smallmatrix}\right)$
      at every point, with eigenvalues $3$ and $1$: strictly convex.
    \item $x^2 - y^2$: \ $H = \operatorname{diag}(2, -2)$ is indefinite: neither convex nor concave.
    \item $-\ln x - \ln y$ on $x, y > 0$: \ $H = \operatorname{diag}(1/x^2, 1/y^2)$: strictly convex.
  \end{itemize}
\end{frame}

\begin{frame}{Critical points of convex functions}
  \begin{theorem}
    Let $f$ be convex and $C^1$ on an open convex set $C$.
    \begin{enumerate}
      \item If $\nabla f(\va) = \mathbf 0$, then $\va$ is a \textbf{global} minimum of $f$ on $C$.
      \item If $f$ is strictly convex, it has at most one global minimum.
    \end{enumerate}
  \end{theorem}
  \pause
  \begin{proof}
    (1) The tangent-plane inequality at $\va$: \ $f(\vy) \ge f(\va) + \mathbf 0 \cdot (\vy - \va) = f(\va)$.
    \pause\\
    (2) If $\va \ne \vb$ were both global minima, with value $m$, then strict convexity at the
    midpoint gives $f\bigl(\frac{\va + \vb}{2}\bigr) < \frac{m + m}{2} = m$, a contradiction.
  \end{proof}
  \pause
  {\small For convex functions, local = global: solving $\nabla f = \mathbf 0$ is the whole problem.\par}
\end{frame}

\begin{frame}{Examples of convex functions}
  \begin{itemize}
    \item Affine functions $\va \cdot \vx + c$: convex and concave ($H = 0$).
    \item $\vx^\top Q \vx$ with $Q$ symmetric positive semidefinite ($H = 2Q$); \ $\norm{\vx}^2$ is
      strictly convex ($H = 2I$).
    \item Every norm, by the triangle inequality:
      $\norm{(1 - t)\vx + t\vy} \le (1 - t)\norm{\vx} + t\norm{\vy}$. \ (Not differentiable at
      $\mathbf 0$ --- convexity does not need derivatives.)
    \item $e^{x + y}$: \ $H = e^{x + y}\begin{pmatrix} 1 & 1 \\ 1 & 1 \end{pmatrix}$, positive
      semidefinite: convex, but not strictly.
  \end{itemize}
  \pause
  \begin{block}{Building convex functions}
    Sums of convex functions with weights $\ge 0$, maxima of convex functions, and
    $\vx \mapsto f(A\vx + \vb)$ for convex $f$ are convex.
  \end{block}
\end{frame}

\begin{frame}{Least squares, finished}
  From Lecture 11: \ $E(\vx) = \norm{A\vx - \vb}^2$ has $\nabla E = 2A^\top A\vx - 2A^\top\vb$, and
  $\nabla E = \mathbf 0$ are the normal equations $A^\top A\vx = A^\top\vb$.
  \pause
  \begin{itemize}
    \item $H_E = 2A^\top A$, and $\vh^\top A^\top A\, \vh = \norm{A\vh}^2 \ge 0$: \ positive
      semidefinite, so $E$ is \textbf{convex}.
    \item Hence \textbf{every} solution of the normal equations is a global minimum.
    \pause
    \item If the columns of $A$ are linearly independent, $A\vh \ne \mathbf 0$ for $\vh \ne \mathbf 0$:
      $H_E$ is positive definite, $E$ is strictly convex, and the minimiser
      $\vx = (A^\top A)^{-1} A^\top \vb$ is unique.
  \end{itemize}
  \pause
  \begin{remark}
    Linear and logistic regression have convex losses; the losses of neural networks are not
    convex, and have many critical points. Next time: gradient descent, analysed on convex
    quadratics.
  \end{remark}
\end{frame}

\section{Summary and practice}

\begin{frame}{Summary}
  \begin{itemize}
    \item \textbf{Fermat}: at an interior local extremum of a differentiable $f$, $\nabla f = \mathbf 0$.
      The converse fails: saddle points.
    \item \textbf{Second-derivative test} at a critical point: $H$ positive definite $\Rightarrow$
      minimum, negative definite $\Rightarrow$ maximum, indefinite $\Rightarrow$ saddle; otherwise
      inconclusive. For $n = 2$: the signs of $D = f_{xx} f_{yy} - f_{xy}^2$ and $f_{xx}$.
    \item \textbf{Global extrema} on a closed bounded set: interior critical points, the boundary,
      compare. A single local minimum need not be global.
    \item \textbf{Convex} $f$: chords above the graph; $H_f$ positive semidefinite. Every critical
      point is a global minimum, and it is unique if $f$ is strictly convex.
    \item Least squares is convex: the normal equations give the global minimum.
  \end{itemize}
\end{frame}

\begin{frame}{Practice}
  \small
  \begin{enumerate}
    \item Find and classify the critical points of $x^3 + y^3 - 3xy$. Is the local minimum global?
    \item Find the minimum of $x^2 + 4y^2 - 2x + 8y + 5$ on $\R^2$. Why is it global?
    \item Find and classify the critical points of $xy\,(1 - x - y)$.
    \item Show that the test is inconclusive for $x^2 + y^4$ and $x^2 - y^4$ at $(0, 0)$, and
      classify $(0, 0)$ for each.
    \item Find the global maximum and minimum of $x^2 - y^2$ on the square $[-1, 1]^2$.
    \item Show that $\norm{\vx}$ is convex on $\R^n$ but not strictly convex.
    \item Show that $\norm{A\vx - \vb}^2$ is strictly convex if the columns of $A$ are linearly
      independent.
  \end{enumerate}
\end{frame}

\begin{frame}{Answers}
  \small
  \begin{enumerate}
    \item $(0, 0)$: $D = -9$, saddle. \ $(1, 1)$: $D = 27$, $f_{xx} = 6$, local minimum $-1$.
      Not global: $f(x, 0) = x^3 \to -\infty$.
    \item $(x - 1)^2 + 4(y + 1)^2$: minimum $0$ at $(1, -1)$. Global: $f$ is convex ($H = \operatorname{diag}(2, 8)$).
    \item Saddles at $(0, 0)$, $(1, 0)$, $(0, 1)$ ($D = -1$); local maximum $\frac1{27}$ at
      $(\frac13, \frac13)$ ($D = \frac13$, $f_{xx} = -\frac23$).
    \item Both have $H = \operatorname{diag}(2, 0)$, $D = 0$. \ $x^2 + y^4$: minimum; \ $x^2 - y^4$: saddle.
    \item Interior: a saddle at $(0, 0)$. Boundary: $1 - y^2$ on $x = \pm 1$, $x^2 - 1$ on $y = \pm 1$.
      Maximum $1$ at $(\pm 1, 0)$, minimum $-1$ at $(0, \pm 1)$.
    \item Triangle inequality. Not strict: for $\vx \ne \mathbf 0$, $\norm{(1 - t)\vx + t\,(2\vx)} = (1 - t)\norm{\vx} + t\norm{2\vx}$.
    \item $\vh^\top H\vh = 2\norm{A\vh}^2 > 0$ for $\vh \ne \mathbf 0$, since $A\vh \ne \mathbf 0$.
  \end{enumerate}
\end{frame}
