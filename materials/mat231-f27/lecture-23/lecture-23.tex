\section{Differential equations}

\begin{frame}{What is a differential equation?}
  An equation relating an unknown function and its derivatives. Its \textbf{order} is the highest
  derivative that appears.
  \pause
  \begin{itemize}
    \item $y' = ky$: \ growth or decay at a rate proportional to the amount.
    \item $P' = rP\,(1 - P/K)$: \ growth limited by a capacity $K$ (the logistic equation).
    \item $T' = -k\,(T - A)$: \ Newton's law of cooling, in a room at temperature $A$.
    \item $m\,x'' = -kx$: \ a mass on a spring (second order, Lecture 24).
    \item $\mathbf x' = A\mathbf x$: \ a system of equations (Lecture 25).
  \end{itemize}
  \pause
  A \textbf{solution} on an interval $I$ is a differentiable function that satisfies the equation at
  every point of $I$. Equations with one independent variable are \textbf{ordinary} (ODEs); with
  several, \textbf{partial} (PDEs).
\end{frame}

\begin{frame}{Initial value problems}
  A first-order \textbf{initial value problem}: \ $y' = f(t, y)$, \ $y(t_0) = y_0$.
  \pause
  \begin{theorem}[Existence and uniqueness]
    If $f$ and $\partial f / \partial y$ are continuous near $(t_0, y_0)$, the problem has exactly one
    solution on some interval around $t_0$.
  \end{theorem}
  \pause
  \begin{itemize}
    \item \textbf{Non-uniqueness} without the hypothesis: $y' = 3y^{2/3}$, $y(0) = 0$ has the solutions
      $y = 0$ and $y = t^3$. Here $\partial f/\partial y = 2y^{-1/3}$ blows up at $y = 0$.
    \item \textbf{Only on some interval}: $y' = y^2$, $y(0) = 1$ has the solution $y = \frac{1}{1 - t}$,
      which blows up at $t = 1$.
  \end{itemize}
\end{frame}

\begin{frame}{Euler's method}
  Follow the direction field in small steps of length $h$:
  \[ t_{k+1} = t_k + h, \qquad y_{k+1} = y_k + h\, f(t_k, y_k) . \]
  \pause
  \begin{example}[$y' = y$, $y(0) = 1$; the exact value $y(1) = e \approx 2.71828$]
    $y_{k+1} = (1 + h)\, y_k$, so the estimate of $y(1)$ is $(1 + h)^{1/h}$:
    \begin{center}
      \begin{tabular}{rll}
        $h = 0.5$: & $1.5^2$ & $= 2.25$ \\
        $h = 0.1$: & $1.1^{10}$ & $\approx 2.5937$ \\
        $h = 0.01$: & $1.01^{100}$ & $\approx 2.7048$
      \end{tabular}
    \end{center}
  \end{example}
  \pause
  {\small The error is roughly proportional to $h$. Gradient descent (Lecture 15) is Euler's method for
  the flow lines of $-\nabla f$, with step $h = \eta$.\par}
\end{frame}

\section{Separable equations}

\begin{frame}{Separation of variables}
  An equation is \textbf{separable} if $y' = g(t)\, h(y)$.
  \pause
  \begin{block}{Method}
    \begin{enumerate}
      \item Constant solutions: every $y_0$ with $h(y_0) = 0$ gives the solution $y = y_0$.
      \item Elsewhere, divide by $h(y)$ and integrate: \
        $\displaystyle \int \frac{dy}{h(y)} = \int g(t)\, dt + C$.
      \item Solve for $y$ if possible, and use the initial condition to find $C$.
    \end{enumerate}
  \end{block}
  \pause
  {\small Why it works: if $H' = 1/h$, then by the chain rule
  $\frac{d}{dt} H(y(t)) = \frac{y'(t)}{h(y(t))} = g(t)$. So $H(y(t)) = \int g\, dt + C$.\par}
\end{frame}

\begin{frame}{Exponential growth and decay}
  $y' = ky$: \ $\displaystyle \int \frac{dy}{y} = \int k\, dt$ \ gives $\ln|y| = kt + C$, so
  \[ y(t) = y_0\, e^{kt} \]
  (this also covers the constant solution $y = 0$).
  \pause
  \begin{itemize}
    \item $k > 0$: growth, doubling time $\frac{\ln 2}{k}$. \ $k < 0$: decay, half-life $\frac{\ln 2}{|k|}$.
  \end{itemize}
  \pause
  \begin{example}[Newton's law of cooling, $T' = -k(T - A)$]
    $T(t) = A + (T_0 - A)\, e^{-kt}$. A cup at $90^\circ$C in a room at $20^\circ$C is at $60^\circ$C
    after $10$ minutes: \ $e^{-10k} = \frac47$. It reaches $40^\circ$C when $e^{-kt} = \frac27$:
    \ $t = 10\, \frac{\ln(7/2)}{\ln(7/4)} \approx 22.4$ minutes.
  \end{example}
\end{frame}

\begin{frame}{The logistic equation}
  \begin{columns}[c]
    \begin{column}{0.58\textwidth}
      $P' = rP\,\bigl(1 - \frac PK\bigr)$, with $r, K > 0$. Separate, and use partial fractions:
      \[ \int \Bigl(\frac1P + \frac{1}{K - P}\Bigr) dP = \int r\, dt , \]
      \pause
      so $\ln\frac{P}{K - P} = rt + C$ and
      \[ P(t) = \frac{K}{1 + A\, e^{-rt}}, \qquad A = \frac{K - P_0}{P_0} . \]
      \pause
      {\small The \textbf{sigmoid} $\sigma(x) = \frac{1}{1 + e^{-x}}$ of logistic regression is the case
      $r = K = 1$: \ $\sigma' = \sigma(1 - \sigma)$.\par}
    \end{column}
    \begin{column}{0.4\textwidth}
      \centering
      \begin{tikzpicture}
        \begin{axis}[width=\linewidth, height=4.4cm, axis lines=left, axis line style={wosm-muted, -},
            xmin=0, xmax=10, ymin=0, ymax=11.5, xtick=\empty, ytick={10}, yticklabels={$K$},
            tick label style={font=\scriptsize, text=wosm-muted}, xlabel={$t$}, label style={font=\scriptsize, text=wosm-muted}]

        \end{axis}

      \par\small Growth slows as $P$ approaches the capacity $K$.
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{A solution can blow up}
  \begin{example}[$y' = t\, y^2$, \ $y(0) = 1$]
    $\displaystyle \int \frac{dy}{y^2} = \int t\, dt$: \ $-\frac1y = \frac{t^2}{2} + C$, and $y(0) = 1$ gives
    $C = -1$. So
    \[ y(t) = \frac{2}{2 - t^2}, \qquad -\sqrt2 < t < \sqrt2 . \]
  \end{example}
  \pause
  \begin{itemize}
    \item The formula makes sense for $|t| > \sqrt2$ too, but that is a different branch: the solution
      of the initial value problem lives on the interval containing $t_0 = 0$.
    \item Nothing in $f(t, y) = ty^2$ warns of this: nonlinear equations can blow up in finite time.
      Linear equations cannot (next).
  \end{itemize}
\end{frame}

\section{First-order linear equations}

\begin{frame}{Linear equations and integrating factors}
  A first-order \textbf{linear} equation: \ $y' + p(t)\, y = q(t)$.
  \pause
  \begin{block}{Integrating factor}
    Let $P$ be an antiderivative of $p$, and $\mu(t) = e^{P(t)}$. Then $\mu' = p\mu$, so
    \[ (\mu y)' = \mu y' + p\mu y = \mu q, \qquad\text{hence}\qquad
       y(t) = \frac{1}{\mu(t)} \Bigl( \int \mu(t)\, q(t)\, dt + C \Bigr) . \]
  \end{block}
  \pause
  \begin{example}[$y' + 2y = e^{-t}$, \ $y(0) = 0$]
    $\mu = e^{2t}$: \ $(e^{2t} y)' = e^{t}$, \ so $e^{2t} y = e^t + C$ and $y = e^{-t} + C e^{-2t}$. \
    $y(0) = 0$ gives $C = -1$: \ $y = e^{-t} - e^{-2t}$.
  \end{example}
\end{frame}

\begin{frame}{Another example}
  \begin{example}[$t\, y' + y = t^2$, \ $t > 0$]
    In standard form $y' + \frac1t\, y = t$: \ $p = \frac1t$, $P = \ln t$, $\mu = t$. Indeed the left side
    of the original equation is already $(ty)'$:
    \[ (ty)' = t^2, \qquad ty = \frac{t^3}{3} + C, \qquad y = \frac{t^2}{3} + \frac Ct . \]
  \end{example}
  \pause
  \begin{itemize}
    \item If $p$ and $q$ are continuous on an interval, the solution exists on the whole interval: linear
      equations do not blow up.
    \item Here $p = \frac1t$ is continuous only for $t > 0$ (or $t < 0$); the solutions with $C \ne 0$ blow up
      at $t = 0$.
  \end{itemize}
\end{frame}

\begin{frame}{Example: a mixing tank}
  \begin{example}[A mixing tank]
    A tank holds $100$ L of water. Brine with $0.5$ kg of salt per litre flows in at $2$ L/min; the
    mixture flows out at $2$ L/min. With $S(t)$ the salt in kg:
    \[ S' = \underbrace{0.5 \cdot 2}_{\text{in}} - \underbrace{\tfrac{S}{100} \cdot 2}_{\text{out}}
       = 1 - \frac{S}{50}, \qquad S(0) = 0 . \]
    \pause
    $\mu = e^{t/50}$, and $S(t) = 50\bigl(1 - e^{-t/50}\bigr)$: the salt approaches $50$ kg, the amount at
    the incoming concentration.
  \end{example}
\end{frame}

\begin{frame}{The structure of the solutions}
  The general solution of $y' + p y = q$ is
  \[ y = y_p + C\, y_h , \]
  where $y_p$ is one particular solution and $y_h = e^{-P(t)}$ solves the homogeneous equation
  $y' + py = 0$.
  \pause
  \begin{itemize}
    \item The map $y \mapsto y' + py$ is \textbf{linear}. So the difference of two solutions of
      $y' + py = q$ solves $y' + py = 0$.
    \item The same structure as for $A\mathbf x = \mathbf b$ in Linear Algebra: one particular solution
      plus the null space.
    \item Superposition: if $y_1$ solves the equation with $q_1$ and $y_2$ with $q_2$, then $y_1 + y_2$
      solves it with $q_1 + q_2$.
  \end{itemize}
  \pause
  {\small Second-order linear equations (Lecture 24) and linear systems (Lecture 25) have the same
  structure, with a null space of dimension $2$, or $n$.\par}
\end{frame}

\section{Summary and practice}

\begin{frame}{Summary}
  \begin{itemize}
    \item An initial value problem $y' = f(t, y)$, $y(t_0) = y_0$ has a unique solution near $t_0$ if $f$
      and $f_y$ are continuous. Without that, uniqueness can fail; solutions can blow up.
    \item Direction fields show solutions at a glance; Euler's method follows them numerically.
    \item \textbf{Separable} $y' = g(t)h(y)$: \ $\int dy/h(y) = \int g\, dt$, plus the constant solutions.
      Growth and decay, cooling, the logistic equation.
    \item \textbf{Linear} $y' + py = q$: multiply by $\mu = e^{\int p}$, then $(\mu y)' = \mu q$.
      Solutions: particular + homogeneous.
  \end{itemize}
\end{frame}

\begin{frame}{Practice}
  \small
  \begin{enumerate}
    \item Solve $y' = -2ty$, $y(0) = 3$.
    \item Solve $y' = 1 + y^2$, $y(0) = 0$. On which interval does the solution exist?
    \item Solve $y' = e^{t - y}$, $y(0) = 1$.
    \item A population follows $P' = P(1 - P/10)$ with $P(0) = 1$. When does it reach $5$?
    \item Solve $y' - y = t$, $y(0) = 0$.
    \item Solve $t\, y' + 2y = 1$ for $t > 0$.
    \item Use Euler's method with $h = 0.25$ to estimate $y(1)$ for $y' = y$, $y(0) = 1$.
    \item Show that $y' = 2\sqrt{|y|}$, $y(0) = 0$ has more than one solution. Which hypothesis of the
      theorem fails?
  \end{enumerate}
\end{frame}

\begin{frame}{Answers}
  \small
  \begin{enumerate}
    \item $\ln|y| = -t^2 + C$: \ $y = 3e^{-t^2}$.
    \item $\arctan y = t$: \ $y = \tan t$, on $(-\frac\pi2, \frac\pi2)$.
    \item $e^y\, dy = e^t\, dt$: \ $e^y = e^t + e - 1$, \ $y = \ln(e^t + e - 1)$.
    \item $P = \frac{10}{1 + 9e^{-t}} = 5$ when $e^{-t} = \frac19$: \ $t = \ln 9 \approx 2.20$.
    \item $\mu = e^{-t}$: \ $(e^{-t}y)' = te^{-t}$, \ $y = -t - 1 + Ce^t$, \ $C = 1$: \ $y = e^t - t - 1$.
    \item $\mu = t^2$: \ $(t^2 y)' = t$, \ $y = \frac12 + \frac{C}{t^2}$.
    \item $1.25^4 \approx 2.441$ \ (against $e \approx 2.718$).
    \item $y = 0$, and $y = t^2$ for $t \ge 0$ (with $y = 0$ for $t < 0$). \ $\partial f / \partial y = 1/\sqrt{y}$
      (for $y > 0$) is unbounded near $y = 0$.
  \end{enumerate}
\end{frame}
